# config.txt for Pi Zero BASIC. Copy this, kernel.img and the firmware files
# (bootcode.bin, start.elf, fixup.dat) to a FAT32 SD card.

# Load our kernel as a 32-bit image at 0x8000.
kernel=kernel.img
arm_64bit=0

# The serial console is the PL011 on GPIO14/15: take it back from Bluetooth,
# and fix its clock so the baud rate is right.
enable_uart=1
dtoverlay=disable-bt
init_uart_clock=48000000

# Full speed from the start (1 GHz on a Zero).
force_turbo=1

# 64 MB for the GPU is plenty for a 640x480 screen; the rest is BASIC's.
gpu_mem=64

# HDMI: the screen size. The firmware reads ONE setting per line, and the
# last one wins, so change the two numbers below; don't add a second pair.
# The text fills the screen in 8x16 cells (set the serial terminal to match):
#   group 2, mode 4    640x480    80x30 text
#   group 2, mode 9    800x600    100x37
#   group 2, mode 16   1024x768   128x48
#   group 2, mode 35   1280x1024  160x64
#   group 1, mode 4    1280x720   160x45
#   group 1, mode 16   1920x1080  240x67
# Delete both lines for the monitor's own size. framebuffer_width and
# framebuffer_height (each on its own line) draw a smaller picture that the
# GPU scales up to the mode: 960 and 540 on a 1080p screen give 120x33 text.
hdmi_group=2
hdmi_mode=4
disable_overscan=1

# HDMI mode, not DVI, even in the monitor timings above: sound goes over HDMI.
hdmi_drive=2
